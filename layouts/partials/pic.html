{{- /* Responsive picture via Hugo's image pipeline. Params (dict): src ("img/x.jpg" — under assets/), alt, sizes (optional), eager (optional: LCP images), vtname (optional view-transition-name), plain (optional: true = full colour on white, specimen plates only; default = duotone via CSS). Missing images warn at build time and render nothing. */ -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $sizes := .sizes | default "(min-width: 1100px) 1100px, 100vw" -}} {{- $vt := .vtname -}} {{- $plain := .plain | default false -}} {{- $eager := .eager | default false -}} {{- $loading := cond $eager "eager" "lazy" -}} {{- $res := resources.Get $src -}} {{- if not $res }}{{ warnf "pic.html: image not found in assets/: %s" $src }}{{ end -}} {{- with $res -}} {{- $img := . -}} {{- $srcset := slice -}} {{- range slice 480 800 1200 -}} {{- if le . $img.Width -}} {{- $r := $img.Process (printf "resize %dx q82" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink .) -}} {{- end -}} {{- end -}} {{- $main := $img -}} {{- if gt $img.Width 1200 }}{{ $main = $img.Process "resize 1200x q82" }}{{ end -}} {{- if $plain -}} {{- else -}} {{- end -}} {{- end -}}
